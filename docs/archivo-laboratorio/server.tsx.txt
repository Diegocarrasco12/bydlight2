// Laboratorio de diseno A/B/C. SOLO DESARROLLO: el layout raiz lo importa unicamente cuando NODE_ENV !== "production".
export type Variant = "a" | "b" | "c" | "d"
export const VARIANTS: { id: Variant; name: string; tag: string }[] = [
  { id: "a", name: "A · Minimalismo tecnológico", tag: "Linear" },
  { id: "b", name: "B · Editorial tecnológico", tag: "Estudio" },
  { id: "c", name: "C · Ingeniería premium", tag: "Vercel" },
  { id: "d", name: "D · Bydlight Signature", tag: "70 % A + 30 % C" },
]

export async function getLab(): Promise<{ variant: Variant | null; fontClass: string }> {
  const v = (await cookies()).get("bd-variant")?.value
  return { variant: v === "a" || v === "b" || v === "c" || v === "d" ? v : null, fontClass: fontClasses }
}
